set(MODERN_IO_PACKAGE_SMOKE_TEST_NAME modern_io_package_consumer_smoke)
set(MODERN_IO_PACKAGE_SMOKE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/consumer")
set(MODERN_IO_PACKAGE_SMOKE_BUILD_DIR "${CMAKE_CURRENT_BINARY_DIR}/consumer-build")
set(MODERN_IO_PACKAGE_SMOKE_INSTALL_PREFIX "${CMAKE_CURRENT_BINARY_DIR}/install-prefix")
set(MODERN_IO_PACKAGE_SMOKE_PACKAGE_DIR "${MODERN_IO_PACKAGE_SMOKE_INSTALL_PREFIX}/${MODERN_IO_INSTALL_CMAKEDIR}")

configure_file(
  "${CMAKE_CURRENT_SOURCE_DIR}/RunPackageSmoke.cmake.in"
  "${CMAKE_CURRENT_BINARY_DIR}/RunPackageSmoke.cmake"
  @ONLY
)

add_test(
  NAME ${MODERN_IO_PACKAGE_SMOKE_TEST_NAME}
  COMMAND "${CMAKE_COMMAND}"
    -DTEST_CONFIG=$<CONFIG>
    -P "${CMAKE_CURRENT_BINARY_DIR}/RunPackageSmoke.cmake"
)

set_tests_properties(${MODERN_IO_PACKAGE_SMOKE_TEST_NAME} PROPERTIES
  LABELS "package"
)
